Cross-isthmian denticle assemblage composition

Overview

Here, we characterize denticle assemblage structure to make comparisons between time periods and regions. This script cleans and filters the raw denticle count data (counts of each functional morphotype), runs the analysis, and plots Figure 2 and the related supplementary figures.

Merge with relevant metadata from denticle count dataframe.

dd_func_pan_age <- dd_func_pan_raw %>%
  #add sed_weight_kg and no_years
  left_join(dd_pan_age_clean_pre[,c(6,9,33)],by="unique") %>% 
  #calculating n demersal ABR and RA denticles
  mutate(abr_dem = abr_tot - abr_pel, 
         rabr_dem = rabr_tot - rabr_pel) %>% 
  mutate(age_group = fct_relevel(age_group,c("pre_exp","modern")))%>%
  #keeping just the relevant samples for each time period
  filter(age_group %in% c("modern","pre_exp")) %>%
  #filtering out one site with difficult to interpret time-averaging
  filter(site_name != "Airport Point") %>%
  #create grouping variable
  unite(sample_group, c("sample","group"),remove=F,sep="_")

Group samples with low N to create more representative samples.

dd_func_pan <- dd_func_pan_age %>% 
  #group together samples with low N
  group_by(basin,region,site_name,sample,sample_group,age_group) %>% 
  #summarize counts per group
  summarize(abr_grouped = sum(abr_tot,na.rm=T),
            defense_grouped = sum(defense,na.rm=T),
            drag_grouped = sum(drag,na.rm=T), 
            gen_func_grouped = sum(gen_func,na.rm=T),
            rabr_grouped = sum(rabr_tot,na.rm=T),
            abr_pel_grouped = sum(abr_pel,na.rm=T),
            rabr_pel_grouped = sum(rabr_pel,na.rm=T),
            abr_dem_grouped = sum(abr_dem,na.rm=T),
            rabr_dem_grouped = sum(rabr_dem,na.rm=T),
            dd_count_grouped = sum(total_dd_count,na.rm=T)) %>% 
  #denticle counts from pelagic & demersal sharks
  mutate(pelagic_count = drag_grouped + abr_pel_grouped + rabr_pel_grouped,
         demersal_count = abr_dem_grouped + rabr_dem_grouped,
         pel_dem_ratio = pelagic_count/demersal_count,
         prop_pelagic = pelagic_count/dd_count_grouped,
         prop_demersal = demersal_count/dd_count_grouped,
  #percent abundance of each functional morphotype
         abr_perc=(abr_grouped/dd_count_grouped)*100,
         defense_perc=(defense_grouped/dd_count_grouped)*100,
         drag_perc=(drag_grouped/dd_count_grouped)*100,
         gen_func_perc=(gen_func_grouped/dd_count_grouped)*100,
         rabr_perc=(rabr_grouped/dd_count_grouped)*100,
         rabr_pel_perc=(rabr_pel_grouped/dd_count_grouped)*100,
         abr_pel_perc=(abr_pel_grouped/dd_count_grouped)*100,
         rabr_dem_perc=(rabr_dem_grouped/dd_count_grouped)*100,
         abr_dem_perc=(abr_dem_grouped/dd_count_grouped)*100) %>%

#Function to extract ellipse data
veganCovEllipse <- function (cov, center = c(0, 0), scale = 1, npoints = 100) 
  {
    theta <- (0:npoints) * 2 * pi/npoints
    Circle <- cbind(cos(theta), sin(theta))
    t(center + scale * t(Circle %*% chol(cov)))
}

#Change taphonomic grades to numerical scores

  mutate(frag_base_score=case_when(
    frag_base=="A" ~ 0,
    frag_base=="B" ~1,
    frag_base=="C"~2)) %>% 
  mutate(frag_crown_score=case_when(
    frag_crown=="A" ~ 0,
    frag_crown=="B" ~1,
    frag_crown=="C"~2,
    frag_crown=="D"~3)) %>% 
  mutate(frag_peaks_score=case_when(
    frag_peaks=="A" ~ 0,
    frag_peaks=="B" ~ 1)) %>% 
  mutate(surf_alt_score=case_when(
    surf_alt=="A" ~ 0,
    surf_alt=="B" ~ 1,
    surf_alt=="C" ~ 2,
    surf_alt=="D" ~ 3)) %>%
  mutate(discolor_score=case_when(
    discolor=="A" ~ 0,
    discolor=="B" ~ 1,
    discolor=="C" ~ 2)) %>%
  mutate(tot_frag_score=(frag_base_score/2)+(frag_crown_score/3)+frag_peaks_score,
         tot_score=tot_frag_score+(surf_alt_score/3)+(discolor_score/2))%>% 
  filter(func_morph != "RAY") #remove any ray denticles
