Cross-isthmian denticle abundances

Overview

Here, we model denticle accumulation rates to make comparisons between time periods and regions. This script cleans and filters the raw denticle count data, runs the analysis, and plots Figure 1 and the related supplementary figures.

Match up denticle counts with age-depth model estimates

dd_pan_age <- dd_pan %>% 
  #combining denticle counts with age estimates
  left_join(bchron_age_depth,by=c("region","sample","depth","unique")) %>% 
  mutate(basin = as.factor(basin),
         region = as.factor(region),
         site_name = as.factor(site_name),
         sample = as.factor(sample)) %>%
  #estimated amount of time in each sample
  mutate(no_years=case_when(
  #bulk sample age estimates
    basin=="Caribbean" ~ num_yrs_est_bulks, 
  #median from age-depth models
    basin=="Pacific" ~ median_num_years_per_sample)) %>%
  #25th percentile, for plotting
  mutate(no_years_25=case_when(
    basin=="Caribbean" ~ num_yrs_est_bulks_lower, 
    basin=="Pacific" ~ perc_25)) %>% 
  #75th percentile, for plotting
  mutate(no_years_75=case_when(
    basin=="Caribbean" ~ num_yrs_est_bulks_upper, 
    basin=="Pacific" ~ perc_75)) %>% 
  #calculate raw denticle accumulation metrics
  mutate(dd_kg=total_dd_count/sed_weight_kg,
         dd_kg_yr=dd_kg/no_years,
         dd_kg_yr_25=dd_kg/no_years_25, 
         dd_kg_yr_75=dd_kg/no_years_75) %>% 
  #using median age for each sampling interval
  rename(age_AD_BC=predicted_age_median, 
         age_BP=predicted_age_BP_median) %>% 
  #filtering out one site with difficult to interpret time-averaging
  filter(site_name != "Airport Point")

Filtering to just core tops/bottoms to facilitate comparability across the isthmus

dd_pan_age_clean_pre <- dd_pan_age %>%
  #applying age cut-offs
  mutate(age_group = case_when(
      age_AD_BC > 1900 ~ "modern", #last ~100 years
      #Holocene, site-specific lower cut-offs given differences in accretion histories, and aiming for similar temporal range across basins
      site_name == "Contadora" & age_AD_BC < -2000 ~ "pre_exp",  
      site_name == "Isla Iguana" & age_AD_BC < -2000 ~ "pre_exp",
      site_name == "Saboga" & age_AD_BC < -1000 ~ "pre_exp", 
      basin == "Caribbean" & type == "modern_bulk" ~ "modern", 
      basin == "Caribbean" & type == "Holocene_bulk" ~ "pre_exp")) %>%
  mutate(age_group = fct_relevel(age_group,c("pre_exp","modern")))%>%
  #keeping just the relevant samples for each time period
  filter(age_group %in% c("modern","pre_exp")) %>%
  droplevels() %>% 
  #adding a column for the sampling year
  mutate(sampling_year = case_when(site_name == "Contadora" ~ 2018,
                                   site_name == "Isla Iguana" ~ 2022,
                                   site_name == "Saboga" ~ 2019,
                                   region == "Bocas del Toro" ~ 2014))

Aggregating core tops into composite “bulk” samples given the presence of looser framework and less compaction. Aim here is to make the top ~15cm usable vs. filtering it out.

dd_pan_age_clean_BULK <- dd_pan_age_clean_pre %>% 
  #filter to the top intervals 0-15cm
  filter(basin=="Pacific" & depth<20) %>%
  group_by(basin, region, site_name,sample) %>% 
  #summarize sample info for these top intervals
  summarize(dd_top = sum(total_dd_count,na.rm=T),
            weight_top = sum(sed_weight_kg,na.rm=T),
            years_top = sampling_year-min(age_AD_BC)) %>% #median date lower sample
  unique() %>% 
  mutate(dd_kg_yr_top = dd_top/weight_top/years_top)
